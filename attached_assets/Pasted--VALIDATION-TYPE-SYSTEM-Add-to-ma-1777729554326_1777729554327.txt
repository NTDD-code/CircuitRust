```typescript
export function getComponentSpec(type: string, model: string) {
  const typeSpec = COMPONENT_REGISTRY[type];
  if (!typeSpec) throw new Error(`Unknown component type: ${type}`);
  const modelSpec = typeSpec.models[model];
  if (!modelSpec) throw new Error(`Unknown model: ${type}::${model}`);
  return modelSpec;
}

export function getPinSpec(type: string, model: string, pin: string) {
  const spec = getComponentSpec(type, model);
  if (!spec.pins || !spec.pins[pin]) {
    throw new Error(`Pin not found: ${type}::${model}.${pin}`);
  }
  return spec.pins[pin];
}

export function getAllComponentTypes(): string[] {
  return Object.keys(COMPONENT_REGISTRY);
}

export function getAllModels(type: string): string[] {
  const typeSpec = COMPONENT_REGISTRY[type];
  if (!typeSpec) return [];
  return Object.keys(typeSpec.models);
}
```

```typescript
export interface ComponentDeclaration {
  id: string;                      // must match /^[A-Z][A-Z0-9_]*$/
  type: string;                    // "Component::Transistor", etc
  model: string;                   // "2N2222", etc
  properties: Record<string, any>; // runtime checked against registry
  lineNumber: number;
  metadata: {
    maxVoltage?: number;
    maxCurrent?: number;
    maxPower?: number;
  };
}

// Net Declaration
export interface NetDeclaration {
  id: string;                      // must match /^[a-z][a-z0-9_]*$/
  type: "VCC" | "GND" | "Signal" | "PWM" | "Analog";
  voltage?: number;
  lineNumber: number;
  connections: PinReference[];
}

// Pin Reference
export interface PinReference {
  componentId: string;
  pinName: string;
  lineNumber: number;
}

// Connection
export interface Connection {
  source: PinReference;
  destination: PinReference;
  lineNumber: number;
  netId?: string;  // assigned during net resolution phase
}

// Complete Circuit AST
export interface CircuitAST {
  name: string;
  components: Map<string, ComponentDeclaration>;
  nets: Map<string, NetDeclaration>;
  connections: Connection[];
  adjacencyList: Map<string, Set<string>>;  // pin → connected pins
  errors: ValidationError[];
  isValid: boolean;
}

  message: string;
  lineNumber: number;
  component?: string;
  pin?: string;
  suggestion: string;
  severity: "error";  // only errors, no warnings
}
```

```typescript
export class StrictCircuitParser {
  parse(sourceCode: string): CircuitAST {
    const ast: CircuitAST = {
      name: "",
      components: new Map(),
      nets: new Map(),
      connections: [],
      adjacencyList: new Map(),
      errors: [],
      isValid: true
    };

    const lines = sourceCode.split("\n");

    // Phase 1: Tokenize and extract declarations
    for (let i = 0; i < lines.length; i++) {
      const line = lines[i].trim();
      const lineNum = i + 1;

      if (!line || line.startsWith("//")) continue;

      // Parse: circuit Name { ... }
      const circuitMatch = line.match(/^circuit\s+(\w+)\s*\{/);
      if (circuitMatch) {
        ast.name = circuitMatch[1];
        continue;
      }

      // Parse: let ID = Component::Type { ... };
      const componentMatch = line.match(
        /let\s+([A-Z][A-Z0-9_]*)\s*=\s*Component::(\w+)\s*\{\s*([^}]+)\}/
      );
      if (componentMatch) {
        const [, id, type, propsStr] = componentMatch;
        const componentType = `Component::${type}`;

        // VALIDATION: Check if component type exists
        if (!COMPONENT_REGISTRY[componentType]) {
          ast.errors.push({
            code: "E001",
            message: `Unknown component type: ${componentType}`,
            lineNumber: lineNum,
            component: id,
            suggestion: `Available types: ${getAllComponentTypes().join(", ")}`,
            severity: "error"
          });
          ast.isValid = false;
          continue;
        }

        // VALIDATION: Check if model exists
        const typeSpec = COMPONENT_REGISTRY[componentType];
        const props = this.parseProperties(propsStr);
        const model = props.model;
        if (!model || !typeSpec.models[model]) {
          ast.errors.push({
            code: "E001",
            message: `Unknown model for ${componentType}: ${model}`,
            lineNumber: lineNum,
            component: id,
            suggestion: `Available models: ${Object.keys(typeSpec.models).join(", ")}`,
            severity: "error"
          });
          ast.isValid = false;
          continue;
        }

        // VALIDATION: Check for duplicate IDs
        if (ast.components.has(id)) {
          ast.errors.push({
            code: "E010",
            message: `Component '${id}' already declared`,
            lineNumber: lineNum,
            component: id,
            suggestion: `Use a different ID like '${id}_2'`,
            severity: "error"
          });
          ast.isValid = false;
          continue;
        }

        // VALIDATION: Validate all properties against registry
        const modelSpec = typeSpec.models[model];
        for (const [propKey, propVal] of Object.entries(props)) {
          if (propKey === "model") continue;  // skip model key itself
          // Type-check numeric properties
          if (typeof propVal === "number") {
            if (modelSpec.maxRatings && modelSpec.maxRatings[propKey]) {
              if (propVal > modelSpec.maxRatings[propKey]) {
                ast.errors.push({
                  code: "E009",
                  message: `Property '${propKey}' exceeds max: ${propVal} > ${modelSpec.maxRatings[propKey]}`,
                  lineNumber: lineNum,
                  component: id,
                  suggestion: `Reduce ${propKey} to ≤ ${modelSpec.maxRatings[propKey]}`,
                  severity: "error"
                });
                ast.isValid = false;
              }
            }
          }
        }

        const decl: ComponentDeclaration = {
          id,
          type: componentType,
          model,
          properties: props,
          lineNumber: lineNum,
          metadata: {
            maxVoltage: modelSpec.maxRatings?.vce_max || props.maxVoltage,
            maxCurrent: modelSpec.maxRatings?.ic_max || props.maxCurrent,
            maxPower: modelSpec.maxRatings?.p_max || props.maxPower
          }
        };
        ast.components.set(id, decl);
        continue;
      }

      // Parse: let NET = Net::Type { ... };
      const netMatch = line.match(
        /let\s+([a-z][a-z0-9_]*)\s*=\s*Net::(\w+)\s*\{([^}]*)\}/
      );
      if (netMatch) {
        const [, id, type, propsStr] = netMatch;
        const props = this.parseProperties(propsStr);

        // VALIDATION: Check duplicate net IDs
        if (ast.nets.has(id)) {
          ast.errors.push({
            code: "E010",
            message: `Net '${id}' already declared`,
            lineNumber: lineNum,
            component: id,
            suggestion: `Use a different ID like '${id}_rail'`,
            severity: "error"
          });
          ast.isValid = false;
          continue;
        }

        const netDecl: NetDeclaration = {
          id,
          type: type as any,
          voltage: props.voltage,
          lineNumber: lineNum,
          connections: []
        };
        ast.nets.set(id, netDecl);
        continue;
      }

      // Parse: connect!(source -> dest);
      const connectMatch = line.match(
        /connect!\(([^→->]+)\s*[-→]\s*([^;)]+)\)/
      );
      if (connectMatch) {
        const [, sourceStr, destStr] = connectMatch;
        const source = this.parsePin(sourceStr.trim(), lineNum, ast);
        const dest = this.parsePin(destStr.trim(), lineNum, ast);

        if (source && dest) {
          ast.connections.push({
            source,
            destination: dest,
            lineNumber: lineNum
          });

          // Build adjacency list for cycle detection
          const sourceKey = `${source.componentId}.${source.pinName}`;
          const destKey = `${dest.componentId}.${dest.pinName}`;
          if (!ast.adjacencyList.has(sourceKey)) {
            ast.adjacencyList.set(sourceKey, new Set());
          }
          ast.adjacencyList.get(sourceKey)!.add(destKey);
        }
      }
    }

    // Phase 2: Post-parse validation rules
    this.runValidationRules(ast);

    return ast;
  }

  private parseProperties(propsStr: string): Record<string, any> {
    const props: Record<string, any> = {};
    // Simple parser: key: value, key: "string", key: 1.5
    const pairs = propsStr.split(",");
    for (const pair of pairs) {
      const [key, val] = pair.split(":").map(s => s.trim());
      if (!key) continue;

      // Parse value: "string" → string, 1.5 → 1.5, true → true
      let parsedVal: any = val;
      if (val.startsWith('"') && val.endsWith('"')) {
        parsedVal = val.slice(1, -1);
      } else if (!isNaN(Number(val))) {
        parsedVal = Number(val);
      } else if (val === "true") {
        parsedVal = true;
      } else if (val === "false") {
        parsedVal = false;
      }

      props[key] = parsedVal;
    }
    return props;
  }

  private parsePin(
    pinStr: string,
    lineNum: number,
    ast: CircuitAST
  ): PinReference | null {
    // Format: Q1.base, vcc (net), gnd (net)
    if (pinStr.includes(".")) {
      const [compId, pinName] = pinStr.split(".");
      // VALIDATION: Component must exist
      if (!ast.components.has(compId)) {
        ast.errors.push({
          code: "E001",
          message: `Undeclared component: '${compId}'`,
          lineNumber: lineNum,
          component: compId,
          suggestion: `Declare it first with: let ${compId} = Component::...`,
          severity: "error"
        });
        ast.isValid = false;
        return null;
      }
      return { componentId: compId, pinName, lineNumber: lineNum };
    } else {
      // Treat as net reference
      if (!ast.nets.has(pinStr)) {
        ast.errors.push({
          code: "E001",
          message: `Undeclared net: '${pinStr}'`,
          lineNumber: lineNum,
          component: pinStr,
          suggestion: `Declare it first with: let ${pinStr} = Net::...`,
          severity: "error"
        });
        ast.isValid = false;
        return null;
      }
      return { componentId: `__net__${pinStr}`, pinName: "vcc", lineNumber: lineNum };
    }
  }

  private runValidationRules(ast: CircuitAST): void {
    // E002: Net connected to itself
    for (const conn of ast.connections) {
      if (
        conn.source.componentId === conn.destination.componentId &&
        conn.source.pinName === conn.destination.pinName
      ) {
        ast.errors.push({
          code: "E002",
          message: `Net connected to itself (short circuit)`,
          lineNumber: conn.lineNumber,
          suggestion: `Remove this redundant connection`,
          severity: "error"
        });
        ast.isValid = false;
      }
    }

    // E003: VCC connected directly to GND
    for (const conn of ast.connections) {
      const isVccToGnd =
        (conn.source.componentId.startsWith("__net__vcc") &&
         conn.destination.componentId.startsWith("__net__gnd")) ||
        (conn.source.componentId.startsWith("__net__gnd") &&
         conn.destination.componentId.startsWith("__net__vcc"));
      if (isVccToGnd) {
        ast.errors.push({
          code: "E003",
          message: `VCC connected directly to GND (catastrophic short)`,
          lineNumber: conn.lineNumber,
          suggestion: `Add a load (resistor, LED, etc) between VCC and GND`,
          severity: "error"
        });
        ast.isValid = false;
      }
    }

    // E005: Floating input pin (declared but never connected)
    for (const [compId, comp] of ast.components) {
      const spec = getComponentSpec(comp.type, comp.model);
      for (const pin in spec.pinTypes) {
        const pinSpec = spec.pinTypes[pin];
        if (pinSpec.type === "input") {
          const pinKey = `${compId}.${pin}`;
          const isConnected = ast.connections.some(
            c => `${c.source.componentId}.${c.source.pinName}` === pinKey ||
                 `${c.destination.componentId}.${c.destination.pinName}` === pinKey
          );
          if (!isConnected) {
            ast.errors.push({
              code: "E005",
              message: `Floating input: ${compId}.${pin} never connected`,
              lineNumber: comp.lineNumber,
              component: compId,
              pin,
              suggestion: `Connect ${compId}.${pin} to a signal net or VCC/GND`,
              severity: "error"
            });
            ast.isValid = false;
          }
        }
      }
    }

    // E006: Component never used
    const usedComponents = new Set<string>();
    for (const conn of ast.connections) {
      usedComponents.add(conn.source.componentId);
      usedComponents.add(conn.destination.componentId);
    }
    for (const [compId] of ast.components) {
      if (!usedComponents.has(compId)) {
        ast.errors.push({
          code: "E006",
          message: `Component '${compId}' declared but never used`,
          lineNumber: ast.components.get(compId)!.lineNumber,
          component: compId,
          suggestion: `Remove this component or connect it`,
          severity: "error"
        });
        ast.isValid = false;
      }
    }

    // E007: LED without series resistor (more complex — see next phase)
    // E008: High-current component without transistor
    // E009: Voltage mismatch
    // E011: Pin connection limit exceeded
    // E012: Missing pull-up/pull-down
    // E013: Capacitor polarity
  }
}
```

```typescript
export async function validateAndInsertAICode(
  generatedCode: string,
  targetEditor: IEditor,
  onError: (errors: ValidationError[]) => void
): Promise<boolean> {
  // Step 1: Strict parsing
  const parser = new StrictCircuitParser();
  const ast = parser.parse(generatedCode);

  // Step 2: If parse failed, show errors and STOP
  if (!ast.isValid || ast.errors.length > 0) {
    console.error("AI generated invalid code:");
    ast.errors.forEach(e => {
      console.error(`[${e.code}] Line ${e.lineNumber}: ${e.message}`);
    });

    // Call error callback (show in UI)
    onError(ast.errors);

    // Optional: Try to fix automatically
    const fixed = await attemptAutoFix(generatedCode, ast.errors);
    if (fixed) {
      return validateAndInsertAICode(fixed, targetEditor, onError);
    }

    return false;  // Reject insertion
  }

  // Step 3: Type checking against registry
  for (const [compId, comp] of ast.components) {
    try {
      const spec = getComponentSpec(comp.type, comp.model);
      // Verify all properties match schema
      for (const [key, val] of Object.entries(comp.properties)) {
        if (key === "model") continue;
        // Type validation here
      }
    } catch (e) {
      onError([{
        code: "E001",
        message: String(e),
        lineNumber: comp.lineNumber,
        component: compId,
        suggestion: "Fix component specification",
        severity: "error"
      }]);
      return false;
    }
  }

  // Step 4: Formal validation (all E* rules)
  const validator = new StrictCircuitValidator();
  const validated = validator.validate(ast);
  if (!validated.isValid) {
    onError(validated.errors);
    return false;
  }

  // Step 5: Safety checks (electrical simulation)
  const safetyCheck = await runSafetyAnalysis(ast);
  if (safetyCheck.hasCriticalIssues) {
    onError([{
      code: "E999",
      message: safetyCheck.issues.join("; "),
      lineNumber: 1,
      suggestion: "Review AI-generated circuit",
      severity: "error"
    }]);
    return false;
  }

  // Step 6: Only if ALL checks pass, insert into editor
  targetEditor.setValue(generatedCode);
  return true;
}

async function attemptAutoFix(
  code: string,
  errors: ValidationError[]
): Promise<string | null> {
```

```typescript
export class StrictCircuitValidator {
  validate(ast: CircuitAST): { isValid: boolean; errors: ValidationError[] } {
    const errors: ValidationError[] = [];

    // E007: LED without series resistor
    errors.push(...this.checkLEDSeries(ast));

    // E008: High-current component without driver
    errors.push(...this.checkHighCurrentDriving(ast));

    // E009: Voltage mismatch
    errors.push(...this.checkVoltageMismatch(ast));

    // E011: Pin connection limits
    errors.push(...this.checkPinConnectionLimits(ast));

    // E012: Missing pull-up/pull-down
    errors.push(...this.checkFloatingPins(ast));

    // E013: Capacitor polarity
    errors.push(...this.checkCapacitorPolarity(ast));

    return {
      isValid: errors.length === 0,
      errors
    };
  }

  private checkLEDSeries(ast: CircuitAST): ValidationError[] {
    const errors: ValidationError[] = [];

    for (const [ledId, ledComp] of ast.components) {
      if (!ledComp.type.includes("LED")) continue;

      // Find all nets connected to this LED's anode
      const anodeConnections = ast.connections.filter(
        c => c.destination.componentId === ledId &&
             c.destination.pinName === "anode"
      );

      for (const conn of anodeConnections) {
        // Trace backwards: anode → net, net → ?
        // If net connects directly to VCC, we need a resistor in between

        // Check if source is VCC
        if (conn.source.componentId.startsWith("__net__vcc")) {
          // Look for a resistor between VCC and this LED
          const hasSeriesResistor = ast.connections.some(
            c => c.destination.componentId === ledId &&
                 c.destination.pinName === "anode" &&
                 // Check if source component is a resistor
                 ast.components.get(c.source.componentId)?.type.includes("Resistor")
          );

          if (!hasSeriesResistor) {
            errors.push({
              code: "E007",
              message: `LED '${ledId}' has no current-limiting resistor in series`,
              lineNumber: ledComp.lineNumber,
              component: ledId,
              suggestion: `Add a resistor between VCC and ${ledId}.anode (e.g., 220Ω for red LED)`,
              severity: "error"
            });
          }
        }
      }
    }

    return errors;
  }

  private checkHighCurrentDriving(ast: CircuitAST): ValidationError[] {
    const errors: ValidationError[] = [];
    const highCurrentTypes = ["Motor", "Buzzer", "Solenoid", "Relay"];

    for (const [compId, comp] of ast.components) {
      const isHighCurrent = highCurrentTypes.some(type =>
        comp.type.includes(type)
      );
      if (!isHighCurrent) continue;

      // Check what's driving this component
      const drivingConnections = ast.connections.filter(
        c => c.destination.componentId === compId &&
             c.destination.pinName === "vcc"
      );

      for (const conn of drivingConnections) {
        // Source must NOT be an MCU GPIO directly
        const sourceComp = ast.components.get(conn.source.componentId);
        if (sourceComp && sourceComp.type.includes("MCU")) {
          // ERROR: MCU driving high-current without transistor/relay
          errors.push({
            code: "E008",
            message: `High-current component '${compId}' driven directly by MCU GPIO ${conn.source.pinName}`,
            lineNumber: conn.lineNumber,
            component: compId,
            suggestion: `Add a transistor or relay between ${conn.source.componentId}.${conn.source.pinName} and ${compId}.vcc`,
            severity: "error"
          });
        }
      }
    }

    return errors;
  }

  private checkVoltageMismatch(ast: CircuitAST): ValidationError[] {
    const errors: ValidationError[] = [];

    // Build voltage map for each net
    const netVoltages = new Map<string, number>();
    for (const [netId, net] of ast.nets) {
      if (net.voltage !== undefined) {
        netVoltages.set(netId, net.voltage);
      }
    }

    // For each component, check pin voltage specs
    for (const [compId, comp] of ast.components) {
      const spec = getComponentSpec(comp.type, comp.model);

      for (const conn of ast.connections) {
        if (conn.destination.componentId !== compId) continue;

        const pinSpec = spec.pinTypes[conn.destination.pinName];
        if (!pinSpec) continue;

        // Get the net voltage
        let netVoltage: number | undefined;
        if (conn.source.componentId.startsWith("__net__")) {
          const netId = conn.source.componentId.replace("__net__", "");
          netVoltage = netVoltages.get(netId);
        }

        // Check if voltage exceeds pin rating
        if (netVoltage !== undefined && spec.maxRatings?.vce_max) {
          if (netVoltage > spec.maxRatings.vce_max) {
            errors.push({
              code: "E009",
              message: `Voltage mismatch: ${compId}.${conn.destination.pinName} rated for ${spec.maxRatings.vce_max}V but net is ${netVoltage}V`,
              lineNumber: conn.lineNumber,
              component: compId,
              suggestion: `Use a component rated for ${netVoltage}V or reduce net voltage`,
              severity: "error"
            });
          }
        }
      }
    }

    return errors;
  }

  private checkPinConnectionLimits(ast: CircuitAST): ValidationError[] {
    const errors: ValidationError[] = [];

    // Count connections per pin
    const pinConnections = new Map<string, number>();
    for (const conn of ast.connections) {
      const srcKey = `${conn.source.componentId}.${conn.source.pinName}`;
      const dstKey = `${conn.destination.componentId}.${conn.destination.pinName}`;
      pinConnections.set(srcKey, (pinConnections.get(srcKey) || 0) + 1);
      pinConnections.set(dstKey, (pinConnections.get(dstKey) || 0) + 1);
    }

    // Check if any pin exceeds its connection limit
    for (const [pinKey, count] of pinConnections) {
      const [compId, pinName] = pinKey.split(".");
      const comp = ast.components.get(compId);
      if (!comp) continue;

      try {
        const spec = getComponentSpec(comp.type, comp.model);
        // Most passive pins can only connect to 2 things (input + output)
        // Check registry for specific limits
        if (count > 2) {
          errors.push({
            code: "E011",
            message: `Pin '${pinKey}' has ${count} connections (max 2)`,
            lineNumber: comp.lineNumber,
            component: compId,
            pin: pinName,
            suggestion: `Use a net to merge connections: all sources → net, net → destination`,
            severity: "error"
          });
        }
      } catch (e) {
        // Component not found
      }
    }

    return errors;
  }
```
